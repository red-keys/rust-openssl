遇到问题：Tongsuo 是一个定制的 OpenSSL 分支，移除了许多算法以符合中国国密标准和安全性要求，而 rust-openssl 期望的是完整的标准 OpenSSL，依赖很多加密算法，被tongsuo移除了的不安全的加密算法。
解决方案是 cargo build --lib， 这样就只生成库而不进行测试，就不会报错某些算法不支持了。

use openssl::ssl::{SslAcceptor, SslFiletype, SslMethod, SslVerifyMode};
use openssl::x509::X509Name;  // Add this import
use std::io::{Read, Write};
use std::net::TcpListener;  // Remove unused TcpStream import
use std::fs::File;
use std::io::BufReader;

fn main() -> std::io::Result<()> {
    // 创建 SSL 接受器
    let mut acceptor = SslAcceptor::mozilla_intermediate(SslMethod::ntls()).unwrap();

        //打开ntls开关
        acceptor.enable_ntls();

    // 设置服务端证书和私钥
    acceptor.set_sign_private_key_file("./ntls_certs/sign_server.key", SslFiletype::PEM).unwrap();
    acceptor.set_sign_certificate_file("./ntls_certs/sign_server.crt", SslFiletype::PEM).unwrap();
    acceptor.check_private_key().unwrap();

    acceptor.set_enc_private_key_file("./ntls_certs/enc_server.key", SslFiletype::PEM).unwrap();
    acceptor.set_enc_certificate_file("./ntls_certs/enc_server.crt", SslFiletype::PEM).unwrap();
    acceptor.check_private_key().unwrap();

    // 配置 NTLS - 要求并验证客户端证书
    acceptor.set_verify(SslVerifyMode::PEER | SslVerifyMode::FAIL_IF_NO_PEER_CERT);

    // 设置信任的 CA 证书
    acceptor.set_ca_file("./ntls_certs/ca.crt").unwrap();

    // 设置客户端 CA 列表 - 使用更简单的方法
    let ca_names = X509Name::load_client_ca_file("./ntls_certs/ca.crt").unwrap();
    acceptor.set_client_ca_list(ca_names);

        // 设置密码套件以及安全级别
        acceptor.set_cipher_list("ECC-SM2-SM4-CBC-SM3")?;
        acceptor.set_security_level(0);

    let acceptor = acceptor.build();

    // 监听端口
    let listener = TcpListener::bind("0.0.0.0:8443")?;

    println!("mTLS Server listening on 0.0.0.0:8443");

    for stream in listener.incoming() {
        match stream {
            Ok(stream) => {
                let acceptor = acceptor.clone();
                std::thread::spawn(move || {
                    match acceptor.accept(stream) {
                        Ok(mut ssl_stream) => {
                            // 获取客户端证书信息
                            if let Some(cert) = ssl_stream.ssl().peer_certificate() {
                                let subject = cert.subject_name();
                                println!("Client connected with certificate: {:?}", subject);
                            }

                            // 读取请求
                            let mut buffer = [0; 1024];
                            ssl_stream.read(&mut buffer).unwrap();
                            println!("Request: {}", String::from_utf8_lossy(&buffer));

                            // 发送响应
                            let response = "HTTP/1.1 200 OK\r\n\r\nHello mTLS Client!";
                            ssl_stream.write_all(response.as_bytes()).unwrap();
                        }
                        Err(e) => {
                            eprintln!("TLS handshake failed: {}", e);
                        }
                    }
                });
            }
            Err(e) => eprintln!("Connection failed: {}", e),
        }
    }

    Ok(())
}

use openssl::ssl::{SslConnector, SslMethod, SslVerifyMode, SslFiletype};  // Add SslFiletype
use std::io::{Read, Write};
use std::net::TcpStream;

fn main() -> std::io::Result<()> {
    // 创建 SSL 连接器
    let mut builder = SslConnector::builder(SslMethod::ntls()).unwrap();

    //打开ntls开关
    builder.enable_ntls();

    // 设置信任的 CA 证书（用于验证服务端）
    builder.set_ca_file("./ntls_certs/ca.crt").unwrap();


    // 设置客户端证书和私钥（用于向服务端证明身份）
    builder.set_sign_private_key_file("./ntls_certs/sign_client.key", SslFiletype::PEM).unwrap();
    builder.set_sign_certificate_file("./ntls_certs/sign_client.crt", SslFiletype::PEM).unwrap();
    builder.check_private_key().unwrap();
    builder.set_enc_private_key_file("./ntls_certs/enc_client.key", SslFiletype::PEM).unwrap();
    builder.set_enc_certificate_file("./ntls_certs/enc_client.crt", SslFiletype::PEM).unwrap();
    builder.check_private_key().unwrap();

    // 启用服务端证书验证
    builder.set_verify(SslVerifyMode::PEER);


    // 设置密码套件以及安全级别
    builder.set_cipher_list("ECC-SM2-SM4-CBC-SM3")?;
    builder.set_security_level(0);

    let connector = builder.build();

    // 连接到服务器
    let stream = TcpStream::connect("localhost:8443")?;
    let mut ssl_stream = connector.connect("server.example.com", stream).unwrap();

    // 验证服务端证书
    if let Some(cert) = ssl_stream.ssl().peer_certificate() {
        let subject = cert.subject_name();
        println!("Connected to server with certificate: {:?}", subject);
    }

    // 发送请求
    let request = "GET / HTTP/1.1\r\nHost: localhost\r\n\r\n";
    ssl_stream.write_all(request.as_bytes()).unwrap();

    // 读取响应
    let mut response = String::new();
    ssl_stream.read_to_string(&mut response).unwrap();
    println!("Response: {}", response);

    Ok(())
}
